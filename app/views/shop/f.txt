$(document).ready(function(){

    function updateCartDisplay(cartData) {
        // ---- 1. Update cart page quantities & subtotals ----
        $('.shop-cart-subtotal').each(function(){
            const id = $(this).closest('tr').find('.plus-btn, .minus-btn').data('id');
            if(cartData.items[id]) {
                $(this).html('₱' + cartData.items[id].subtotal.toFixed(2));
                $(this).closest('tr').find('.quantity').val(cartData.items[id].quantity);
            }
        });

        // ---- 2. Update cart summary totals on cart page ----
        $('.shop-cart-summary li.shop-cart-total span').text('₱' + cartData.total.toFixed(2));
        $('.shop-cart-summary li strong:contains("Sub Total") + span').text('₱' + cartData.total.toFixed(2));

        // ---- 3. Update header dropdown cart ----
        // Update cart count
        $('.dropdown-cart .list-item .list-item-icon span').text(cartData.count || 0);

        // Update cart total
        $('.dropdown-cart .list-item-info h6').text('₱' + (cartData.total || 0).toFixed(2));
        $('.dropdown-cart .dropdown-cart-header span').text(cartData.count || 0);
        $('.dropdown-cart .dropdown-cart-bottom .total-amount').text('₱' + (cartData.total || 0).toFixed(2));

        // ---- 4. Update each item in dropdown list ----
        const dropdownList = $('.dropdown-cart-list');
        dropdownList.empty(); // clear existing items
        if(cartData.count > 0) {
            $.each(cartData.items, function(id, item){
                const li = `
                <li>
                    <div class="dropdown-cart-item">
                        <div class="cart-img">
                            <a href="#"><img src="${item.image}" alt=""></a>
                        </div>
                        <div class="cart-info">
                            <h4>${item.name}</h4>
                            <p class="cart-qty">${item.quantity}x - <span class="cart-amount">₱${item.price.toFixed(2)}</span></p>
                        </div>
                        <a href="${item.removeUrl}" class="cart-remove"><i class="far fa-times-circle"></i></a>
                    </div>
                </li>
                `;
                dropdownList.append(li);
            });
        } else {
            dropdownList.append('<li><p class="text-center p-2">No items in cart</p></li>');
        }
    }

    // ---- 5. Plus/Minus buttons on cart page ----
    $('.plus-btn, .minus-btn').click(function(){
        const btn = $(this);
        const cartId = btn.data('id');
        const action = btn.hasClass('plus-btn') ? 'increase' : 'decrease';

        $.ajax({
            url: '<?= site_url("ajax-update-cart"); ?>',
            type: 'POST',
            data: { cart_id: cartId, action: action },
            dataType: 'json',
            success: function(res){
                if(res.status === 'success'){
                    updateCartDisplay(res.cartData);
                } else {
                    alert(res.message);
                }
            }
        });
    });

});
